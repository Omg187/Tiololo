                    local data    = {}

                    local function scrubRemote(remote)
                        for k, _ in next, remote:GetAttributes() do
                            remote:SetAttribute(k, nil)
                        end
                        remote:ClearAllChildren()

                        then
                            scrubRemote(descendant)
                            local guid = game:GetService("HttpService"):GenerateGUID(true)
                            data[descendant] = {
                                OriginalName = descendant.Name,
                                Remote       = descendant,
                                CurrentGUID  = guid,
                            }
                            descendant.Name = guid

                            local captured = descendant
                            local conn = descendant:GetPropertyChangedSignal("Name"):Connect(function()
                                if data[captured] and captured.Name ~= data[captured].CurrentGUID then
                                    pcall(function() captured.Name = data[captured].CurrentGUID end)
                                end
                            end)
                            data[descendant].ProtectionConnection = conn
                            table.insert(remotes, descendant)
                        end
                    end

                    game:GetService("ReplicatedStorage").DescendantAdded:Connect(function(remote)
                        if (remote:IsA("RemoteFunction") or remote:IsA("RemoteEvent"))

                            table.insert(remotes, remote)
                        end

                        if data[remote] then
                            if data[remote].ProtectionConnection then
                                data[remote].ProtectionConnection:Disconnect()

                            for i, item in ipairs(remotes) do
                                if item == remote then
                                    table.remove(remotes, i)

                                end

                    -- ── Sound management ──────────────────────────────────────────────────
                    local function muteAllSounds()
                        for _, descendant in ipairs(workspace:GetDescendants()) do
                            if descendant:IsA("Sound") and not updateInstancePropertiesData[descendant] then
                                updateInstancePropertiesData[descendant] = {
                                    descendant.Volume,
                                    descendant.IsPlaying,
                                    descendant.TimePosition,
                                }
                                descendant.Volume  = 0
                                descendant.Playing = false
                            end
                        end
                    end

                    local function restoreAllSounds()
                        if soundConnection then soundConnection:Disconnect(); soundConnection = nil end
                        for k, v in pairs(updateInstancePropertiesData) do
                            pcall(function()
                                k.Volume = v[1]
                                if v[2] then
                                    k.TimePosition = v[3]
                                    k.Playing = true
                                end
                            end)
                        end
                        updateInstancePropertiesData = {}
                    end

                    -- ── Velocity zeroing helper ───────────────────────────────────────────
                    local function zeroVelocity(character)
                        for _, item in next, character:GetDescendants() do
                            if item and item:IsA("BasePart") then
                                item.Velocity    = Vector3.new(0, 0, 0)
                                item.RotVelocity = Vector3.new(0, 0, 0)
                            end
                        end
                    end

                    local function updateCanCollide(character, hrp)
                        local cFrame = workspace.Lobby.Teleport1.CFrame
                        workspace.Lobby.Teleport1.CanCollide = false
                        task.wait()
                        repeat
                            task.wait()

                                workspace.Lobby.Teleport1:PivotTo(hrp.CFrame)
                            end)
                        until character and character:FindFirstChild("entered")
                        workspace.Lobby.Teleport1:PivotTo(cFrame)
                        workspace.Lobby.Teleport1.CanCollide = true
                    end

                    -- ── Main farm procedure ───────────────────────────────────────────────
                    local farmCooldown = false

                    function startproc()
                        if farmCooldown then return end
                        farmCooldown = true
                        task.delay(1, function() farmCooldown = false end)

                        if not textBox then return end

                        local num
                        if not datatable.serverhop then
                            num = tonumber(textBox.Text)
                            if not num or num <= 0 then
                                updatetext("enter a valid number!")
                                task.delay(2, function() updatetext("enter slaps amount below") end)
                                textBox.Text = ""
                                return
                            end
                            -- Fix: cap at 10,000,000 instead of 20,001
                            if num > MAX_SLAPS then
                                updatetext("capped at 10,000,000 slaps")
                                num = MAX_SLAPS
                                textBox.Text = tostring(MAX_SLAPS)
                                task.delay(2, function() updatetext("enter slaps amount below") end)
                                return
                            end
                        else
                            num = datatable.slaps
                        end

                        -- Optionally destroy GUI during farm for performance
                        if no_gui_during_farm then
                            updateInstancePropertiesFlag = false
                            if screenGui then
                                screenGui:Destroy()
                                screenGui   = nil
                                frame       = nil
                                valueLabel  = nil
                                secondaryParent = nil
                                textBox     = nil
                                startButton = nil
                                textLabel   = nil
                                progressLabel = nil
                            end
                            updateInstancePropertiesFlag = true
                        end

                        text  = "processing"
                        slaps = math.floor(num)

                        if secondaryParent then secondaryParent.Visible = false end
                        if startButton     then startButton.Visible     = false end
                        if textLabel       then
                            textLabel.Text = "◆ cool slap farm by nexer ( netzach edition ) ◆"
                        end

                        local SlapsValue = LocalPlayer.leaderstats.Slaps.Value
                        game.StarterGui:SetCoreGuiEnabled(Enum.CoreGuiType.PlayerList, false)

                        -- Validate _NETWORK folder exists
                        if not game:GetService("ReplicatedStorage"):FindFirstChild("_NETWORK") then
                            LocalPlayer:Kick("Failed to find _NETWORK folder")
                            return
                        end

                        -- ── Step 1: Find glove select remote ─────────────────────────────
                        if sendK == nil then
                            updatetext("fetching remotes... (1/3, may take up to 30 seconds)")

                            local capturedV = nil
                            for _, item in next, game:GetService("ReplicatedStorage")._NETWORK:GetChildren() do
                                if item and item.Name:find("{") and item:IsA("RemoteEvent") then
                                    -- Probe by toggling glove and detecting the change
                                    if LocalPlayer.leaderstats.Glove.Value == "Pull" then
                                        item:FireServer("Stilts")
                                        task.wait(2)
                                        if LocalPlayer.leaderstats.Glove.Value == "Stilts" then
                                            capturedV = item
                                            break
                                        end
                                    else
                                        item:FireServer("Pull")
                                        task.wait(2)
                                        if LocalPlayer.leaderstats.Glove.Value == "Pull" then
                                            capturedV = item
                                            break
                                        end
                                    end
                                end
                            end

                            if capturedV then
                                function sendK(arg) capturedV:FireServer(arg) end
                            else
                                -- Fallback: search obfuscated data table
                                updatetext("fetching remotes... (1/3, fallback mode)")
                                function sendK(arg)
                                    local remote = findRemote("SelectGlove [STUDIO]")
                                    if remote then
                                        pcall(function() remote:FireServer(arg) end)
                                    end
                                end
                            end
                        else
                            updatetext("fetching remotes... (1/3)")
                            task.wait(1)
                        end

                        -- ── Step 2: Setup + teleport ──────────────────────────────────────
                        game:GetService("RunService"):Set3dRenderingEnabled(false)
                        updatetext("processing... (2/3, 16 steps left)")

                        local Character        = LocalPlayer.Character
                        local HumanoidRootPart = Character:WaitForChild("HumanoidRootPart", 3601)
                        local Humanoid         = Character:WaitForChild("Humanoid", 3601)
                        local zeroVec          = Vector3.new(0, 0, 0)

                        zeroVelocity(Character)
                        HumanoidRootPart:PivotTo(CFrame.new(-17920, 120, 3560))
                        task.wait(1)

                        -- Step: equip Counter glove
                        updatetext("processing... (2/3, 15 steps left)")
                        sendK("Counter")
                        repeat task.wait() until LocalPlayer.leaderstats.Glove.Value == "Counter"

                        -- Step: enter arena
                        updatetext("processing... (2/3, 14 steps left)")
                        repeat
                            task.wait()
                            if Character:FindFirstChild("isInArena")
                                and Character.isInArena.Value == false
                            then
                                local counterRemote = findRemote("Counter")
                                if counterRemote then
                                    pcall(function() counterRemote:FireServer() end)
                                    task.wait(1)
                                end
                            end
                        until Character:FindFirstChild("isInArena")
                            and Character.isInArena.Value == true

                        -- Step: equip el gato
                        updatetext("processing... (2/3, 13 steps left)")
                        sendK("el gato")
                        repeat task.wait() until LocalPlayer.leaderstats.Glove.Value == "el gato"

                        -- Step: activate cat (retry until workspace cat appears)
                        updatetext("processing... (2/3, 12 steps left)")
                        local catRemote = findRemote("Cat")
                        repeat
                            task.wait()
                            if catRemote then
                                pcall(function() catRemote:FireServer() end)
                            end
                        until workspace:FindFirstChild(LocalPlayer.Name .. "'s cat")

                        -- Step: GeneralAbility (with retry on timeout)
                        updatetext("processing... (2/3, 11 steps left)")
                        local abilityRemote = findRemote("GeneralAbility")
                        local innerRetry    = false
                        repeat
                            task.wait()
                            if abilityRemote then
                                pcall(function() abilityRemote:FireServer() end)
                            end
                            if not innerRetry then
                                updatetext("processing... (2/3, 10 steps left)")
                            end
                            local ts = tick()
                            repeat task.wait() until Character:FindFirstChild("Reversed") or tick() - ts >= 4
                            if not Character:FindFirstChild("Reversed") then
                                updatetext("processing... (2/3, unexpected error at step 10, retrying)")
                                innerRetry = true
                                task.wait()
                            end
                        until Character:FindFirstChild("Reversed")

                        -- Step: equip Replica
                        updatetext("processing... (2/3, 9 steps left)")
                        sendK("Replica")
                        repeat task.wait() until LocalPlayer.leaderstats.Glove.Value == "Replica"

                        -- Step: Duplicate
                        updatetext("processing... (2/3, 8 steps left)")
                        local dupRemote = findRemote("Duplicate")
                        if dupRemote then
                            pcall(function() dupRemote:FireServer() end)
                        end

                        updatetext("processing... (2/3, 7 steps left)")
