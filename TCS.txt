local function new(vector, newNumber)
	if typeof(vector) == "Vector3" then
		return Vector3.new(vector.X * newNumber, vector.Y, vector.Z * newNumber)
	end

	return vector
end
local function handleCondition(k)
	local conditionFlag = not k

	if not conditionFlag then
		conditionFlag = type(k) ~= "string"
	end

	if conditionFlag then
		return false
	end

	local lower = k:lower()
	local find = lower:find("pass")

	if not find then
		find = lower:find("lob")

		if not find then
			find = lower:find("cross")
		end
	end

	return find
end

local function secondaryNew(vector, newNumber)
	if typeof(vector) == "Vector3" then
		return Vector3.new(
			math.clamp(vector.X * newNumber, -150, 150),
			math.clamp(vector.Y * newNumber, -150, 150),
			math.clamp(vector.Z * newNumber, -150, 150)
		)
	end

	return vector
end
local function handleOption(item)
	local optionFlag = not item

	if not optionFlag then
		optionFlag = type(item) ~= "string"
	end

	if optionFlag then
		return false
	end

	local lower = item:lower()

	if handleCondition(lower) then
		return false
	end

	for k, _ in pairs(optionData) do
		local searchableText = k

		if lower == searchableText or lower:find(searchableText) then
			return true
		end
	end

	return false
end
local function handler(data, text)
	if not text then
		text = ""
	end

	for k, item in pairs(data) do
		local secondaryK = k
		local option = type(secondaryK) == "string" and secondaryK:lower() or ""
		local secondaryText = text

		if option ~= "" then
			secondaryText = text .. " " .. option
		end

		if typeof(item) == "Vector3" then
			local curveEnabled = curveEnabledData[option]

			if curveEnabled then
				curveEnabled = loopTpData.CurveEnabled
			end

			if curveEnabled then
				data[secondaryK] = secondaryNew(item, loopTpData.CurveMult)
			else
				local powerEnabled = powerEnabledData[option] or option == ""

				if powerEnabled then
					powerEnabled = loopTpData.PowerEnabled
				end

				if powerEnabled then
					data[secondaryK] = new(item, loopTpData.PowerMult)
				end
			end
		elseif type(item) == "number" then
			local powerEnabled = powerEnabledData[option]

			if powerEnabled then
				powerEnabled = loopTpData.PowerEnabled
			end

			if powerEnabled then
				data[secondaryK] = item * loopTpData.PowerMult
			end
		elseif type(item) == "table" then
			handler(item, secondaryText)
		end
	end
end

local function handleSearchableText()
	local descendants = {}
	for index, item in ipairs(Workspace:GetDescendants()) do
		local isName = item:IsA("BasePart")

		if isName then
			isName = item.Name == "TPS"
		end

		if isName then
			table.insert(descendants, item)
		end
	end
	for index, item in ipairs(Workspace:GetDescendants()) do
		local isName = item:IsA("Model")

		if isName then
			isName = item.Name == "TPS"
		end

		if isName then
			for _, descendant in ipairs(item:GetDescendants()) do
				if descendant:IsA("BasePart") then
					table.insert(descendants, descendant)
				end
			end
		end
	end
	local vData = {}
	local vs = {}
	for _, item in ipairs(descendants) do
		if not vData[item] then
			vData[item] = true
			table.insert(vs, item)
		end
	end

	return vs
end

secondaryGetrawmetatable.__newindex = newcclosure(function(argument, secondaryArgument, vector)
	if typeof(vector) == "Vector3" then
		local Character = LocalPlayer.Character

		if Character then
			Character = typeof(argument) == "Instance" and argument:IsDescendantOf(Character)
		end

		if Character then
			return newindex(argument, secondaryArgument, vector)
		end

		local parent = getcallingscript and getcallingscript()
		local conditionOption = parent and parent.Name:lower() or ""

		if parent then
			local Parent = parent.Parent

			if Parent then
				Parent = parent.Parent.Name:lower()
			end

			parent = Parent
		end

		local k = parent or ""

		if handleCondition(conditionOption) or handleCondition(k) then
			return newindex(argument, secondaryArgument, vector)
		end

		local condition = handleOption(conditionOption)

		if not condition then
			condition = handleOption(k)

			if not condition then
				condition = lowerData[conditionOption] or tick() - 0 < 0.8
			end
		end

		if condition then
			local PowerEnabled = loopTpData.PowerEnabled

			if PowerEnabled then
				PowerEnabled = secondaryArgument == "Velocity"

				if not PowerEnabled then
					PowerEnabled = secondaryArgument == "AssemblyLinearVelocity"
						or secondaryArgument == "VectorVelocity"
				end
			end

			if PowerEnabled then
				local PowerMult = loopTpData.PowerMult

				vector = if typeof(vector) ~= "Vector3"
					then vector
					else Vector3.new(vector.X * PowerMult, vector.Y, vector.Z * PowerMult)
			end

			if loopTpData.CurveEnabled then
				local flag = secondaryArgument == "AssemblyAngularVelocity"

				if not flag then
					flag = secondaryArgument == "RotVelocity"

					if not flag then
						flag = secondaryArgument == "AngularVelocity" and argument:IsA("AngularVelocity")
					end
				end

				if flag then
					vector = secondaryNew(vector, loopTpData.CurveMult)
				else
					local flag = secondaryArgument == "Force"

					if flag then
						flag = argument:IsA("BodyForce") or argument:IsA("VectorForce")
					end

					if not flag then
						flag = secondaryArgument == "Thrust"

						if flag then
							flag = argument:IsA("BodyThrust")
						end
					end

					if flag then
						local CurveMult = loopTpData.CurveMult

						vector = if typeof(vector) ~= "Vector3" then vector else vector * CurveMult
					else
						local flag = secondaryArgument == "Torque"

						if flag then
							flag = argument:IsA("Torque")
						end

						if flag then
							vector = secondaryNew(vector, loopTpData.CurveMult)
						end
					end
				end
			end
		end
	end

	return newindex(argument, secondaryArgument, vector)
end)
